using UnityEngine;
using Unity.Cinemachine; // Lub 'using Cinemachine;' w zależności od wersji

public class AssignCameraBounds : MonoBehaviour
{
    void Start()
    {
        // Szukamy komponentu graniczącego w załadowanej kamerze
        CinemachineConfiner2D confiner = FindObjectOfType<CinemachineConfiner2D>();

        if (confiner != null)
        {
            // Przypinamy ten konkretny Polygon Collider do kamery
            PolygonCollider2D myCollider = GetComponent<PolygonCollider2D>();
            confiner.BoundingShape2D = myCollider;

            // Odświeżamy cache, żeby kamera nie pamiętała starej mapy
            confiner.InvalidateBoundingShapeCache();
        }
    }
}

using UnityEngine;

public class AudioManager : MonoBehaviour
{
    public static AudioManager instance;

    [Header("Źródła Dźwięku (Przypisz w Inspectorze)")]
    public AudioSource musicSource; // Kanał muzyki
    public AudioSource sfxSource;   // Kanał efektów

    // Zmienne do trzymania głośności
    private float musicVolume = 0.5f;
    private float sfxVolume = 0.5f;

    void Awake()
    {
        if (instance == null)
        {
            instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Destroy(gameObject);
        }
    }

    public void StopMusic()
    {
        musicSource.Stop();
    }

    // --- FUNKCJE DLA SUWAKÓW ---

    public void SetMusicVolume(float volume)
    {
        musicVolume = volume;
        musicSource.volume = musicVolume;
    }

    public void SetSFXVolume(float volume)
    {
        sfxVolume = volume;
        sfxSource.volume = sfxVolume;
    }

    // --- ODTWARZANIE MUZYKI (TŁO) ---
    public void PlayMusic(AudioClip clip)
    {
        if (clip == null) return;

        // Sprytny warunek: Jeśli ta piosenka już gra, nie resetuj jej!
        if (musicSource.clip == clip && musicSource.isPlaying) return;

        musicSource.clip = clip;
        musicSource.loop = true; // Muzyka zawsze w pętli
        musicSource.volume = musicVolume; // Ustaw głośność suwaka
        musicSource.Play();
    }

    // --- ODTWARZANIE EFEKTÓW (JEDNORAZOWE) ---
    // Tego używasz do kroków, ataków, kliknięć
    public void PlaySFX(AudioClip clip)
    {
        if (clip == null) return;

        // PlayOneShot pozwala na nakładanie się dźwięków (np. wybuch i krzyk naraz)
        sfxSource.PlayOneShot(clip, sfxVolume);
    }
    public float GetMusicVolume()
    {
        return musicVolume;
    }

    public float GetSFXVolume()
    {
        return sfxVolume;
    }
}

using UnityEngine;
using UnityEngine.UI; // WAŻNE: Wymagane do obsługi Text i Slider
using TMPro;
using System.Collections.Generic; //lista

public class BattleHUD : MonoBehaviour
{
    public TextMeshProUGUI nameText;
    public TextMeshProUGUI levelText;
    public Slider hpSlider;

    // 1. NOWE: Pole na tekst z liczbami HP
    public TextMeshProUGUI hpText;

    // 2. NOWE: Zmienna prywatna do zapamiętania Max HP
    private int _maxHP;

    [Header("Drużyna (Jajka)")]
    public Image[] teamEggs; // Tablica 6 obrazków reprezentujących jajka
    public Color aliveColor = new Color(1f, 1f, 1f, 1f); // Zwykły (Biały)
    public Color faintedColor = new Color(0.3f, 0.3f, 0.3f, 1f); // Ciemny szary (Zemdlał)

    // Ta funkcja ustawi UI na samym początku walki
    public void SetHUD(BattleUnit unit)
    {
        nameText.text = unit.unitName;
        levelText.text = "Lvl " + unit.unitLevel;

        hpSlider.maxValue = unit.maxHP;
        hpSlider.value = unit.currentHP;

        // 3. NOWE: Zapamiętujemy MaxHP i ustawiamy tekst startowy
        _maxHP = unit.maxHP;
        hpText.text = unit.currentHP + " / " + _maxHP;
    }

    public void SetHP(int hp)
    {
        hpSlider.value = hp;

        // 4. NOWE: Aktualizujemy tekst używając nowej wartości i zapamiętanego Max
        hpText.text = hp + " / " + _maxHP;
    }

    // --- NOWA FUNKCJA: Aktualizacja Ikon Jajek ---
    public void UpdateTeamUI(List<Pokezaur> team)
    {
        // Sprawdzamy wszystkie 6 miejsc na jajka
        for (int i = 0; i < teamEggs.Length; i++)
        {
            if (i < team.Count)
            {
                // Mamy Pokezaura na tym slocie
                teamEggs[i].gameObject.SetActive(true); // Pokazujemy jajko

                if (team[i].currentHP > 0)
                {
                    teamEggs[i].color = aliveColor; // Żyje = jasne
                }
                else
                {
                    teamEggs[i].color = faintedColor; // Nie żyje = ciemne
                }
            }
            else
            {
                // Nie mamy Pokezaura na tym slocie (np. mamy tylko 3, a to jest slot 4)
                teamEggs[i].gameObject.SetActive(false); // Ukrywamy jajko
            }
        }
    }


}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using TMPro;
using UnityEngine.SceneManagement;

public enum BattleState { START, PLAYERTURN, ENEMYTURN, WON, LOST }

public class BattleSystem : MonoBehaviour
{
    public BattleState state;

    // Jednostki na scenie
    public BattleUnit playerUnit;
    public BattleUnit enemyUnit;

    // Interfejsy (HUD)
    public BattleHUD playerHUD;
    public BattleHUD enemyHUD;

    // Źródło danych gracza (ScriptableObject)
    public PlayerData playerDataSource;

    // Tekst dialogowy
    public TextMeshProUGUI dialogueText;

    [Header("Panel zmiany Pokezaura w walce")]
    public GameObject partySwitchPanel;       // Panel z 6 przyciskami drużyny
    public Button[] partySlotButtons;         // 6 przycisków slotów
    public TextMeshProUGUI[] partySlotTexts;  // Teksty na 6 przyciskach (nazwa + HP)
    public Image[] partySlotIcons;            // NOWE: 6 obrazków (Icon) wewnątrz przycisków
    public GameObject cancelSwitchButton;     // Przycisk "Anuluj" (ukrywany, gdy zmiana jest wymuszona po śmierci)

    [Header("Debug / Testowanie")]
    public EnemyData debugEnemy;

    [Header("Dźwięki")]
    public AudioClip victorySound;
    public AudioClip defeatSound;
    public AudioClip hitSound;

    // Śledzimy dokładnie tego Pokezaura, który aktualnie stoi na arenie
    private Pokezaur activePokezaur;
    private bool isForcedSwitch = false;

    void Start()
    {
        if (partySwitchPanel != null)
            partySwitchPanel.SetActive(false);

        state = BattleState.START;
        StartCoroutine(SetupBattle());
    }

    IEnumerator SetupBattle()
    {
        EnemyData enemyData = BattleTransitionManager.enemyToFight;

        if (enemyData == null)
        {
            if (debugEnemy != null)
            {
                Debug.LogWarning("TRYB TESTOWY: Używam przeciwnika z pola DebugEnemy.");
                enemyData = debugEnemy;
                BattleTransitionManager.enemyToFight = debugEnemy;
            }
            else
            {
                Debug.LogError("BŁĄD: Brak przeciwnika! Odpal grę od Mapy albo przypisz DebugEnemy.");
                dialogueText.text = "Błąd: Brak danych wroga!";
                yield break;
            }
        }

        enemyUnit.SetupEnemy(enemyData);

        // Szukamy pierwszego żywego Pokezaura i zapisujemy go w polu klasy activePokezaur
        activePokezaur = GetFirstAlivePokezaur();

        if (activePokezaur != null)
        {
            playerUnit.SetupPlayer(activePokezaur);
        }
        else
        {
            dialogueText.text = "Nie masz Pokezaurów zdolnych do walki!";
            yield return WaitForInputOrTime(1.5f);

            state = BattleState.LOST;
            EndBattle();
            yield break;
        }

        enemyHUD.SetHUD(enemyUnit);
        playerHUD.SetHUD(playerUnit);
        playerHUD.UpdateTeamUI(playerDataSource.team);

        dialogueText.text = "Dzikie " + enemyUnit.unitName + " atakuje!";
        yield return WaitForInputOrTime(0.5f);

        state = BattleState.PLAYERTURN;
        PlayerTurn();
    }

    private Pokezaur GetFirstAlivePokezaur()
    {
        foreach (Pokezaur poke in playerDataSource.team)
        {
            if (poke.currentHP > 0)
                return poke;
        }
        return null;
    }

    void PlayerTurn()
    {
        dialogueText.text = "Wybierz akcję:";
    }

    // --- SYSTEM ZMIANY POKEZAURA W WALCE ---

    // Podepnij pod przycisk "DRUŻYNA / ZMIEŃ" w menu walki
    public void OnOpenPartyButton()
    {
        if (state != BattleState.PLAYERTURN) return;

        isForcedSwitch = false;
        OpenPartySwitchPanel();
    }

    private void OpenPartySwitchPanel()
    {
        if (partySwitchPanel == null) return;

        partySwitchPanel.SetActive(true);

        // Jeśli zmiana jest wymuszona (po omdleniu), nie pozwalamy zamknąć okna przyciskiem "Anuluj"
        if (cancelSwitchButton != null)
            cancelSwitchButton.SetActive(!isForcedSwitch);

        // Odświeżamy wygląd 6 przycisków
        for (int i = 0; i < partySlotButtons.Length; i++)
        {
            if (i < playerDataSource.team.Count)
            {
                Pokezaur poke = playerDataSource.team[i];
                partySlotButtons[i].gameObject.SetActive(true);

                int displayHP = (poke == activePokezaur) ? playerUnit.currentHP : poke.currentHP;
                string statusTag = (poke == activePokezaur) ? " [WALCZY]" : (displayHP <= 0 ? " [ZEMDLAŁ]" : "");

                // Wyświetlamy Nazwę, Poziom, HP oraz statystyki ATK i DEF
                partySlotTexts[i].text = $"<b>{poke.pokeName}</b> (Lv.{poke.level}){statusTag}\n" +
                                         $"HP: {displayHP}/{poke.maxHP} | Siła: {poke.damage} | Obr: {poke.defense}";

                // Wyświetlamy obrazek (przód) Pokezaura na przycisku
                if (partySlotIcons != null && i < partySlotIcons.Length && partySlotIcons[i] != null)
                {
                    partySlotIcons[i].enabled = (poke.sprite != null);
                    partySlotIcons[i].sprite = poke.sprite;
                }

                partySlotButtons[i].interactable = (displayHP > 0 && poke != activePokezaur);
            }
            else
            {
                partySlotButtons[i].gameObject.SetActive(false);
            }
        }
    }

    // Podepnij pod przycisk "Anuluj" w panelu drużyny
    public void OnClosePartyButton()
    {
        if (isForcedSwitch) return; // Nie można anulować, gdy obecny Pokezaur zemdlał
        if (partySwitchPanel != null)
            partySwitchPanel.SetActive(false);
    }

    // Podepnij pod 6 przycisków slotów (z argumentem int od 0 do 5)
    public void OnSelectPartyMember(int slotIndex)
    {
        if (slotIndex < 0 || slotIndex >= playerDataSource.team.Count) return;

        Pokezaur chosenPoke = playerDataSource.team[slotIndex];
        if (chosenPoke.currentHP <= 0 || chosenPoke == activePokezaur) return;

        if (partySwitchPanel != null)
            partySwitchPanel.SetActive(false);

        StartCoroutine(SwitchPokezaurRoutine(chosenPoke));
    }

    IEnumerator SwitchPokezaurRoutine(Pokezaur newPokezaur)
    {
        state = BattleState.ENEMYTURN; // Blokujemy przyciski na czas animacji/komunikatu

        // 1. Zapisujemy HP dotychczasowego Pokezaura (jeśli jeszcze żyje)
        if (activePokezaur != null)
        {
            activePokezaur.currentHP = playerUnit.currentHP;
        }

        // 2. Ustawiamy nowego Pokezaura jako aktywnego
        activePokezaur = newPokezaur;
        playerUnit.SetupPlayer(activePokezaur);
        playerHUD.SetHUD(playerUnit);
        playerHUD.UpdateTeamUI(playerDataSource.team);

        dialogueText.text = $"Do boju, {activePokezaur.pokeName}!";
        yield return WaitForInputOrTime(1f);

        // 3. Jeśli to była zmiana wymuszona po śmierci, gracz dostaje swoją turę.
        // Jeśli dobrowolna w trakcie tury — przechodzimy do tury wroga.
        if (isForcedSwitch)
        {
            isForcedSwitch = false;
            state = BattleState.PLAYERTURN;
            PlayerTurn();
        }
        else
        {
            StartCoroutine(EnemyTurn());
        }
    }

    // --- FUNKCJE PRZYCISKÓW AKCJI ---

    public void OnAttackButton()
    {
        if (state != BattleState.PLAYERTURN) return;
        StartCoroutine(PlayerAttack());
    }

    public void OnHealButton()
    {
        if (state != BattleState.PLAYERTURN) return;
        StartCoroutine(PlayerHeal());
    }

    public void OnPotionButton()
    {
        if (state != BattleState.PLAYERTURN) return;

        ItemData potionToUse = null;
        foreach (InventorySlot slot in playerDataSource.inventory)
        {
            if (slot.item.itemType == ItemType.HealHP)
            {
                potionToUse = slot.item;
                break;
            }
        }

        if (potionToUse != null)
        {
            StartCoroutine(UseItemRoutine(potionToUse));
        }
        else
        {
            dialogueText.text = "Nie masz żadnych mikstur w plecaku!";
        }
    }

    public void OnRunButton()
    {
        if (state != BattleState.PLAYERTURN) return;
        StartCoroutine(TryToRun());
    }

    IEnumerator TryToRun()
    {
        state = BattleState.ENEMYTURN;
        dialogueText.text = "Próbujesz uciec...";
        yield return WaitForInputOrTime(1f);

        bool success = Random.Range(0, 2) == 0;

        if (success)
        {
            dialogueText.text = "Udało się uciec!";
            yield return WaitForInputOrTime(1f);

            // POPRAWKA: Zapisujemy HP dokładnie tego Pokezaura, który aktualnie walczył
            if (activePokezaur != null)
            {
                activePokezaur.currentHP = playerUnit.currentHP;
            }

            BattleTransitionManager.isReturningFromBattle = true;
            BattleTransitionManager.justRanAway = true;

            SceneManager.LoadScene("MapScene");
        }
        else
        {
            dialogueText.text = "Nie udało się uciec!";
            yield return WaitForInputOrTime(1f);

            dialogueText.text = "Tracisz turę!";
            yield return WaitForInputOrTime(1f);

            StartCoroutine(EnemyTurn());
        }
    }

    IEnumerator UseItemRoutine(ItemData itemToUse)
    {
        state = BattleState.ENEMYTURN;

        playerDataSource.RemoveItem(itemToUse);
        dialogueText.text = $"Użyto: {itemToUse.itemName}!";
        yield return WaitForInputOrTime(1f);

        if (itemToUse.itemType == ItemType.CatchItem)
        {
            dialogueText.text = "Rzucasz Jajem Pokezaura!";
            yield return WaitForInputOrTime(1f);

            if (Random.value <= 0.5f)
            {
                dialogueText.text = "Udało się! Pokezaur złapany!";
                yield return WaitForInputOrTime(1.5f);

                // POPRAWKA: Zapisujemy aktualne HP naszego walczącego Pokezaura przed wyjściem na mapę!
                if (activePokezaur != null)
                {
                    activePokezaur.currentHP = playerUnit.currentHP;
                }

                if (playerDataSource.team.Count < 6)
                {
                    Pokezaur newPoke = new Pokezaur(BattleTransitionManager.enemyToFight);
                    playerDataSource.team.Add(newPoke);
                    dialogueText.text = $"{newPoke.pokeName} dołącza do drużyny!";
                }
                else
                {
                    dialogueText.text = "Brak miejsca w drużynie! (System PC wkrótce)";
                }

                yield return WaitForInputOrTime(1.5f);

                BattleTransitionManager.justRanAway = false;
                BattleTransitionManager.defeatedEnemies.Add(BattleTransitionManager.currentEnemyID);
                SceneManager.LoadScene("MapScene");
                yield break;
            }
            else
            {
                dialogueText.text = "O nie! Pokezaur się uwolnił!";
                yield return WaitForInputOrTime(1f);
            }
        }
        else if (itemToUse.itemType == ItemType.HealHP)
        {
            playerUnit.Heal(itemToUse.value);
            playerHUD.SetHP(playerUnit.currentHP);

            // POPRAWKA: Zapisujemy uleczone HP do aktywnie walczącego Pokezaura (a nie sztywno do team[0])
            if (activePokezaur != null)
            {
                activePokezaur.currentHP = playerUnit.currentHP;
            }

            dialogueText.text = $"+{itemToUse.value} HP!";
            yield return WaitForInputOrTime(1f);
        }

        StartCoroutine(EnemyTurn());
    }

    // --- LOGIKA WALKI ---

    IEnumerator PlayerAttack()
    {
        state = BattleState.ENEMYTURN;

        int dmg = playerUnit.CalculateDamage();
        bool isDead = enemyUnit.TakeDamage(dmg);
        enemyHUD.SetHP(enemyUnit.currentHP);

        dialogueText.text = "Zadajesz " + dmg + " pkt obrażeń!";

        if (hitSound != null && AudioManager.instance != null)
        {
            AudioManager.instance.PlaySFX(hitSound);
        }

        yield return WaitForInputOrTime(0.5f);

        if (isDead)
        {
            state = BattleState.WON;
            EndBattle();
        }
        else
        {
            StartCoroutine(EnemyTurn());
        }
    }

    IEnumerator PlayerHeal()
    {
        state = BattleState.ENEMYTURN;

        int healAmount = (playerUnit.unitLevel * 2) + 6;
        int minHeal = Mathf.FloorToInt(healAmount * 0.9f);
        int maxHeal = Mathf.FloorToInt(healAmount * 1.1f);
        if (minHeal < 1) minHeal = 1;

        int healRandom = Random.Range(minHeal, maxHeal + 1);

        playerUnit.Heal(healRandom);
        playerHUD.SetHP(playerUnit.currentHP);

        // POPRAWKA: Synchronizujemy HP z obiektem aktywnego Pokezaura
        if (activePokezaur != null)
        {
            activePokezaur.currentHP = playerUnit.currentHP;
        }

        dialogueText.text = "Odzyskujesz " + healRandom + " HP!";
        yield return WaitForInputOrTime(0.5f);

        StartCoroutine(EnemyTurn());
    }

    IEnumerator EnemyTurn()
    {
        dialogueText.text = enemyUnit.unitName + " atakuje!";
        yield return WaitForInputOrTime(0.5f);

        int dmg = enemyUnit.CalculateDamage();
        bool isDead = playerUnit.TakeDamage(dmg);
        playerHUD.SetHP(playerUnit.currentHP);

        // Zapisujemy aktualne HP w obiekcie naszego Pokezaura
        if (activePokezaur != null)
        {
            activePokezaur.currentHP = playerUnit.currentHP;
        }

        dialogueText.text = enemyUnit.unitName + " zadał Ci " + dmg + " obrażeń!";
        yield return WaitForInputOrTime(0.5f);

        if (isDead)
        {
            // Odświeżamy ikonki jajek (jedno jajko staje się szare/martwe)
            playerHUD.UpdateTeamUI(playerDataSource.team);

            dialogueText.text = $"{activePokezaur.pokeName} pada nieprzytomny!";
            yield return WaitForInputOrTime(1.5f);

            // Sprawdzamy, czy mamy jeszcze jakiegokolwiek żywego Pokezaura w drużynie
            Pokezaur nextAlive = GetFirstAlivePokezaur();

            if (nextAlive != null)
            {
                // Jeśli mamy podpięty panel wyboru drużyny, otwieramy go w trybie wymuszonym
                if (partySwitchPanel != null)
                {
                    dialogueText.text = "Wybierz kolejnego Pokezaura!";
                    isForcedSwitch = true;
                    OpenPartySwitchPanel();
                }
                else
                {
                    // Zabezpieczenie: jeśli nie zbudowałeś jeszcze panelu UI, automatycznie wypuszcza kolejnego żywego!
                    isForcedSwitch = true;
                    yield return StartCoroutine(SwitchPokezaurRoutine(nextAlive));
                }
            }
            else
            {
                // Dopiero gdy WSZYSTKIE Pokezaury w drużynie mają 0 HP -> przegrywamy walkę
                state = BattleState.LOST;
                EndBattle();
            }
        }
        else
        {
            state = BattleState.PLAYERTURN;
            PlayerTurn();
        }
    }

    // --- KONIEC WALKI ---

    void EndBattle()
    {
        StartCoroutine(EndBattleRoutine());
    }

    IEnumerator EndBattleRoutine()
    {
        if (state == BattleState.WON)
        {
            if (AudioManager.instance != null)
                AudioManager.instance.StopMusic();

            if (victorySound != null && AudioManager.instance != null)
                AudioManager.instance.PlaySFX(victorySound);

            dialogueText.text = "Wygrałeś walkę!";
            yield return WaitForInputOrTime(1.5f);

            int xpReward = BattleTransitionManager.enemyToFight.xpReward;
            int goldReward = BattleTransitionManager.enemyToFight.goldReward;
            playerDataSource.AddGold(goldReward);

            dialogueText.text = "Zdobywasz " + goldReward + " Złota!";
            yield return WaitForInputOrTime(1.5f);

            if (activePokezaur != null)
            {
                // Najpierw zapisujemy HP, z jakim skończył walkę
                activePokezaur.currentHP = playerUnit.currentHP;

                dialogueText.text = activePokezaur.pokeName + " zdobywa " + xpReward + " XP!";
                yield return WaitForInputOrTime(1.5f);

                bool leveledUp = activePokezaur.AddXP(xpReward);

                if (leveledUp)
                {
                    // POPRAWKA: Odświeżamy całe playerUnit i HUD (nowe maxHP, nowy level i pełne HP)
                    playerUnit.SetupPlayer(activePokezaur);
                    playerHUD.SetHUD(playerUnit);

                    dialogueText.text = "AWANS! Poziom " + activePokezaur.level + "!";
                    yield return WaitForInputOrTime(1.5f);
                    dialogueText.text = "Statystyki wzrosły!";
                    yield return WaitForInputOrTime(1f);
                }
            }

            BattleTransitionManager.justRanAway = false;
            BattleTransitionManager.defeatedEnemies.Add(BattleTransitionManager.currentEnemyID);
            SceneManager.LoadScene("MapScene");
        }
        else if (state == BattleState.LOST)
        {
            if (AudioManager.instance != null)
                AudioManager.instance.StopMusic();

            if (defeatSound != null && AudioManager.instance != null)
                AudioManager.instance.PlaySFX(defeatSound);

            int lostGold = playerDataSource.LoseGoldOnDeath();
            dialogueText.text = "Zostałeś pokonany...";
            yield return WaitForInputOrTime(1f);
            dialogueText.text = $"Straciłeś {lostGold} sztuk złota.";
            yield return WaitForInputOrTime(1f);

            BattleTransitionManager.isReturningFromBattle = false;

            if (playerDataSource.team.Count > 0)
            {
                foreach (Pokezaur p in playerDataSource.team)
                {
                    p.currentHP = p.maxHP;
                }
            }

            SceneManager.LoadScene("MapScene");
        }
    }

    IEnumerator WaitForInputOrTime(float duration)
    {
        float timer = 0f;
        while (timer < duration)
        {
            if (Input.GetMouseButtonDown(0) || Input.GetKeyDown(KeyCode.Space))
            {
                yield break;
            }
            timer += Time.deltaTime;
            yield return null;
        }
    }
}

using System.Collections.Generic;
using UnityEngine;

public static class BattleTransitionManager
{
    //WALKA
    public static EnemyData enemyToFight;

    // NOWE: Tu zapiszemy pozycję gracza
    public static Vector3 lastPlayerPosition;

    // NOWE: Przełącznik, czy wracamy z walki
    public static bool isReturningFromBattle = false;

    public void SetupEnemy(EnemyData enemyData)
    {
        unitName = enemyData.enemyName;
        maxHP = enemyData.maxHealth;
        currentHP = enemyData.maxHealth; // Przeciwnik zawsze zaczyna z pełnym życiem
        damage = enemyData.attackPower;
        unitLevel = enemyData.unitLevel;
        defense = enemyData.defense;

        // Ustawiamy grafikę potwora

        unitLevel = myPokezaur.level;
        maxHP = myPokezaur.maxHP;
        damage = myPokezaur.damage;
        defense = myPokezaur.defense;
        currentHP = myPokezaur.currentHP;

        mySpriteRenderer.sprite = myPokezaur.backSprite != null ? myPokezaur.backSprite : myPokezaur.sprite;
    }

    // Metoda zwraca 'true', jeśli jednostka umarła
    public bool TakeDamage(int rawDamage)
    {
        // Redukujemy obrażenia o obronę
        int actualDamage = rawDamage - defense/2;

        // Zabezpieczenie: Żeby atak nie leczył (jeśli obrona jest wyższa niż atak)
        // Zawsze zadajemy minimum 1 obrażenia (chyba że chcesz całkowitą odporność)
        if (actualDamage < 1) actualDamage = 1;

        currentHP -= actualDamage;

        if (currentHP <= 0)
        {
            currentHP = 0;
            return true;
        }
        return false;
    }

    public void Heal(int amount)
    {
        currentHP += amount;
        if (currentHP > maxHP)
        {
            currentHP = maxHP; // Nie można mieć więcej niż max życia
        }
    }

    // Nowa funkcja do losowania siły ataku
    public int CalculateDamage()
    {
        // Przykład: Obrażenia mogą być o 2 mniejsze lub o 2 większe od bazy
        // Jeśli damage = 10, wylosuje liczbę z przedziału 8-12
        int minDmg = Mathf.FloorToInt(damage * 0.9f);
        int maxDmg = Mathf.FloorToInt(damage * 1.1f);

        // Zabezpieczenie: żeby nie zadać 0 lub ujemnych obrażeń
        if (minDmg < 1) minDmg = 1;

        // Random.Range dla int jest ekskluzywne dla górnej granicy, więc dajemy +1
        return Random.Range(minDmg, maxDmg + 1);
    }
}

    }

            {
                xpSlider.maxValue = leadPoke.xpToNextLevel;
                xpSlider.value = leadPoke.currentXP;
            }
            if (xpText != null)
            {
                xpText.text = $"{leadPoke.currentXP} / {leadPoke.xpToNextLevel} XP";
            }

public enum InteractionType { Sign, Item, NPC, Shop, Trainer }

public class Interactable : MonoBehaviour
{
    [Header("Typ Interakcji")]
    public InteractionType type;

    [Header("Unikalne ID (Generuje się samo)")]
    public string itemID; // <-- NOWOŚĆ: Do zapisu stanu zebrania

    [Header("Opis (Dla UI)")]
    public string objectName;

    [Header("Dla Znaków / NPC")]
    [TextArea] public string messageContent;

    [Header("Dla Przedmiotów")]
    public ItemData itemToGive;
    public PlayerData playerData;
    public AudioClip interactSound;

    [Header("Dla Sklepu")]
    public List<ItemData> shopInventory;

    [Header("Dla Trenerów")]
    public EnemyData enemy;
    public string battleSceneName = "BattleScene";

    private bool isPlayerInRange = false;
    private bool justInteracted = false;

    // --- NOWOŚĆ: Sprawdzanie przy starcie ---
    void Start()
    {
        // Jeśli to przedmiot i już go zebraliśmy -> zniszcz go od razu
        if (type == InteractionType.Item && playerData != null)
        {
            if (playerData.CheckIfCollected(itemID))
            {
                Destroy(gameObject);
            }
        }
    }

    void Update()
    {
        if (isPlayerInRange && !justInteracted && Input.GetKeyDown(KeyCode.E))
        {
            PerformAction();
        }
    }

    void PerformAction()
    {
        // 1. Dźwięk
        if (interactSound != null && AudioManager.instance != null)
        {
            AudioManager.instance.PlaySFX(interactSound);
        }

        // 2. Logika
        switch (type)
        {
            case InteractionType.Sign:
                GameNotifications.instance.ShowMessage(messageContent, 4f);
                break;

            case InteractionType.Item:
                PickUpItem();
                break;

            case InteractionType.NPC:
                // Tu możesz dodać logikę dialogu
                break;
            case InteractionType.Shop:
                if (ShopMenu.instance != null)
                {
                    ShopMenu.instance.OpenShop(shopInventory);
                }
                break;
            case InteractionType.Trainer:
                StartSpecificBattle();
                break;
        }
    }

    void PickUpItem()
    {
        if (itemToGive != null && playerData != null)
        {
            justInteracted = true;

            // Dodaj do plecaka
            playerData.AddItem(itemToGive);

            // --- NOWOŚĆ: Oznaczamy jako zebrane w PlayerData ---
            playerData.MarkAsCollected(itemID);

            // Wyświetl komunikat
            GameNotifications.instance.ShowMessage($"Podniesiono: {itemToGive.itemName}", 3f);

            Destroy(gameObject);
        }
    }

    // --- FIZYKA ---

    private void OnTriggerEnter2D(Collider2D other)
    {
        if (other.CompareTag("Player"))
        {
            isPlayerInRange = true;

            if (type == InteractionType.Sign)
            {
                PerformAction();
            }
            else
            {
                string prompt = "";

                switch (type)
                {
                    case InteractionType.Item:
                        // Używamy objectName, który automatycznie pobrał się z ItemData
                        prompt = $"[E] Podnieś: {objectName}";
                        break;
                    case InteractionType.NPC:
                        prompt = $"[E] Rozmawiaj: {objectName}";
                        break;
                    case InteractionType.Shop:
                        prompt = $"[E] Handluj: {objectName}";
                        break;
                    case InteractionType.Trainer:
                        prompt = $"[E] Walcz z {objectName}";
                        break;
                }

[CreateAssetMenu(fileName = "New Player", menuName = "Pokemon/Player")]
public class PlayerData : ScriptableObject
{
    public string unitName;

    [Header("Drużyna (Max 6)")]
    public List<Pokezaur> team = new List<Pokezaur>();

    [Header("Statystyki Trenera")]
    public float moveSpeed = 5f;
    public int currentGold = 0;


    [Header("Ustawienia")]
    public bool allowChinesePropaganda = false;


    [Header("Audio")]
    public AudioClip EQSound;
    public AudioClip potionSound;
    public AudioClip consumableSound;

    [Header("Zapis Gry")]
    public List<string> collectedItemIDs = new List<string>();

    public List<InventorySlot> inventory = new List<InventorySlot>();
    public ItemData currentEquippedItem; // Zostawiamy dla pierwszego stworka

    public bool CheckIfCollected(string id) { return collectedItemIDs.Contains(id); }
    public void MarkAsCollected(string id) { if (!collectedItemIDs.Contains(id)) collectedItemIDs.Add(id); }

    public void AddItem(ItemData itemToAdd)
    {
        foreach (InventorySlot slot in inventory)
        {
            if (slot.item == itemToAdd)
            {
                slot.amount++;
                return;
            }
        }
        inventory.Add(new InventorySlot(itemToAdd, 1));
    }

    public void RemoveItem(ItemData itemToRemove)
    {
        foreach (InventorySlot slot in inventory)
        {
            if (slot.item == itemToRemove)
            {
                slot.amount--;
                if (slot.amount <= 0) inventory.Remove(slot);
                return;
            }
        }
    }

    public void AddGold(int amount) { currentGold += amount; }

    public bool SpendGold(int amount)
    {
        if (currentGold >= amount) { currentGold -= amount; return true; }
        return false;
    }

    public int LoseGoldOnDeath()
    {
        int goldToLose = Mathf.FloorToInt(currentGold * 0.25f);
        currentGold -= goldToLose;
        if (currentGold < 0) currentGold = 0;
        return goldToLose;
    }

    // --- UŻYWANIE PRZEDMIOTÓW (NA LIDERZE DRUŻYNY) ---
    public void UseItem(ItemData item, int targetIndex = 0)
    {
        if (team.Count == 0 && item.itemType != ItemType.Gold)
        {
            Debug.LogWarning("Brak Pokezaurów w drużynie!");
            return;
        }

        if (targetIndex < 0 || targetIndex >= team.Count) targetIndex = 0;
        Pokezaur targetPoke = (team.Count > 0) ? team[targetIndex] : null;

        switch (item.itemType)
        {
            case ItemType.HealHP:
            case ItemType.Consumable:
                if (targetPoke.currentHP >= targetPoke.maxHP) return;
                targetPoke.currentHP += item.value;
                if (targetPoke.currentHP > targetPoke.maxHP) targetPoke.currentHP = targetPoke.maxHP;
                RemoveItem(item);
                if (item.itemType == ItemType.HealHP) PlaySound(potionSound);
                else PlaySound(consumableSound);
                break;

            case ItemType.Gold:
                AddGold(item.value);
                RemoveItem(item);
                break;

            case ItemType.PermSTR:
                targetPoke.damage += item.value;
                RemoveItem(item);
                PlaySound(potionSound);
                break;

            case ItemType.PermDEF:
                targetPoke.defense += item.value;
                RemoveItem(item);
                PlaySound(potionSound);
                break;

            case ItemType.PermHP:
                targetPoke.maxHP += item.value;
                targetPoke.currentHP += item.value;
                RemoveItem(item);
                PlaySound(potionSound);
                break;

            case ItemType.Equipment:
                // Ekwipunek zawsze zakładamy na aktualnego lidera (team[0])
                HandleEquipment(item, team[0]);
                PlaySound(EQSound);
                break;
        }
    }

    public void SwapLeader(int newLeaderIndex)
    {
        if (newLeaderIndex <= 0 || newLeaderIndex >= team.Count) return;

        // Jeśli lider ma założony przedmiot, zdejmujemy go przed zamianą i zakładamy nowemu liderowi
        ItemData equipped = currentEquippedItem;
        if (equipped != null) UnequipItem(team[0]);

        Pokezaur temp = team[0];
        team[0] = team[newLeaderIndex];
        team[newLeaderIndex] = temp;

        if (equipped != null) EquipItem(equipped, team[0]);
    }


    void PlaySound(AudioClip clip)
    {
        if (clip != null && AudioManager.instance != null) AudioManager.instance.PlaySFX(clip);
    }

    void HandleEquipment(ItemData item, Pokezaur poke)
    {
        if (currentEquippedItem == item) UnequipItem(poke);
        else
        {
            if (currentEquippedItem != null) UnequipItem(poke);
            EquipItem(item, poke);
        }
    }

    void EquipItem(ItemData item, Pokezaur poke)
    {
        currentEquippedItem = item;
        foreach (StatModifier mod in item.modifiers)
        {
            switch (mod.statType)
            {
                case StatType.Strength: poke.damage += mod.amount; break;
                case StatType.Defense: poke.defense += mod.amount; break;
                case StatType.MaxHP: poke.maxHP += mod.amount; poke.currentHP += mod.amount; break;
                case StatType.Speed: moveSpeed += mod.amount; break; // Speed zostaje dla trenera
            }
        }
    }

    void UnequipItem(Pokezaur poke)
    {
        if (currentEquippedItem != null)
        {
            foreach (StatModifier mod in currentEquippedItem.modifiers)
            {
                switch (mod.statType)
                {
                    case StatType.Strength: poke.damage -= mod.amount; break;
                    case StatType.Defense: poke.defense -= mod.amount; break;
                    case StatType.MaxHP:
                        poke.maxHP -= mod.amount;
                        if (poke.currentHP > poke.maxHP) poke.currentHP = poke.maxHP;
                        break;
                    case StatType.Speed: moveSpeed -= mod.amount; break;
                }
            }
            currentEquippedItem = null;
        }
    }
}

            // Jeśli to mapa - włączamy gracza
            GetComponent<SpriteRenderer>().enabled = true;
            if (rb != null) rb.simulated = true;
        }

        // 2. LOGIKA MAPY (Wykonuje się TYLKO jeśli to NIE jest BattleScene)
        GameObject grassObj = GameObject.Find("Enemy_Grass");
        if (grassObj != null)
        {
            grassTilemap = grassObj.GetComponent<Tilemap>();
        }

        if (BattleTransitionManager.isReturningFromBattle)
        {
            transform.position = BattleTransitionManager.lastPlayerPosition;

            // --- NOWOŚĆ: Przywracanie warstwy wizualnej ---
            if (BattleTransitionManager.isOnElevation)
            {
                GetComponent<SpriteRenderer>().sortingOrder = 10;
            }
            else
            {
                GetComponent<SpriteRenderer>().sortingOrder = 3; // lub Twoja domyślna wartość
            }
            // ----------------------------------------------

            // Logika ochrony
            if (BattleTransitionManager.justRanAway)
            {
                currentStepTracker = -6f;
            }

            // Resetujemy flagi DOPIERO PO powrocie na mapę
            BattleTransitionManager.justRanAway = false;
            BattleTransitionManager.isReturningFromBattle = false;
        }
    }


    void Update()
    {
        // 1. BLOKADA DIALOGU
        if (DialogueManager.instance != null && DialogueManager.instance.isDialogueActive)
        {
            rb.linearVelocity = Vector2.zero; // Zatrzymanie fizyki
            anim.SetFloat("Speed", 0);
            return; // Nic więcej nie robimy
        }

        // 2. ODCZYT KLAWISZY
        movement.x = Input.GetAxisRaw("Horizontal");
        movement.y = Input.GetAxisRaw("Vertical");

        // 3. SPRAWDZENIE CZY SIĘ RUSZAMY
        if (movement.sqrMagnitude > 0.01f)
        {
            isMoving = true;

            // Animacje
            anim.SetFloat("Horizontal", movement.x);
            anim.SetFloat("Vertical", movement.y);
            anim.SetFloat("Speed", 1);

            // --- A. LICZENIE KROKÓW DO WALKI (Z nową prędkością!) ---
            // TU JEST ZMIANA: Mnożymy przez playerData.moveSpeed zamiast lokalnego moveSpeed
            currentStepTracker += playerData.moveSpeed * Time.deltaTime;

            if (currentStepTracker >= stepLength)
            {
                currentStepTracker = 0; // Reset licznika
                CheckForEncounters();   // Sprawdź czy walka
            }

            // --- B. DŹWIĘK KROKÓW (TUPANIE) ---
            if (Time.time >= nextStepTime)
            {
                if (stepSound != null && AudioManager.instance != null)
                {
                    AudioManager.instance.PlaySFX(stepSound);

                    // Im szybciej biegniesz, tym częściej tupiesz? 
                    // Możesz zostawić stałe stepRate albo zrobić dynamiczne: stepRate / (playerData.moveSpeed / 5f)
                    nextStepTime = Time.time + stepRate;
                }
            }
        }
        else
        {
            // Jeśli stoimy w miejscu
            isMoving = false;
            anim.SetFloat("Speed", 0);
        }
    }

    void FixedUpdate()
    {
        rb.linearVelocity = movement.normalized * playerData.moveSpeed;
    }

    void CheckForEncounters()
    {
        if (grassTilemap == null) return;

        Vector3Int gridPosition = grassTilemap.WorldToCell(transform.position);

        if (grassTilemap.HasTile(gridPosition))
        {
            int roll = Random.Range(0, 1001);

            if (roll <= encounterChance)
            {
                BattleTransitionManager.lastPlayerPosition = transform.position;
                BattleTransitionManager.isReturningFromBattle = true;

                if (potentialEnemies.Length > 0)
                {
                    int randomIndex = Random.Range(0, potentialEnemies.Length);
                    BattleTransitionManager.enemyToFight = potentialEnemies[randomIndex];

                    movement = Vector2.zero;
                    if (rb != null) rb.linearVelocity = Vector2.zero;
                    SceneManager.LoadScene(battleSceneName);
                }
            }
        }
    }
}

        pokeName = wildEnemy.enemyName;
        level = wildEnemy.unitLevel;
        maxHP = wildEnemy.maxHealth;
        currentHP = wildEnemy.maxHealth;
        damage = wildEnemy.attackPower;
        defense = wildEnemy.defense;
        sprite = wildEnemy.battleSprite;
        backSprite = wildEnemy.backSprite != null ? wildEnemy.backSprite : wildEnemy.battleSprite;

        currentXP = 0;
        xpToNextLevel = 100 * level;
    }

    // Logika awansu przeniesiona z PlayerData
    public bool AddXP(int amount)
    {
        currentXP += amount;
        bool hasLeveledUp = false;

        while (currentXP >= xpToNextLevel)
        {
            LevelUp();
            hasLeveledUp = true;
        }

        return hasLeveledUp;
    }

    private void LevelUp()
    {
        level++;
        currentXP -= xpToNextLevel;
        xpToNextLevel = Mathf.RoundToInt(xpToNextLevel * 1.5f);

        maxHP += 5;
        currentHP = maxHP; // Leczenie przy awansie
        damage += 2;
        defense += 1;

        Debug.Log($"{pokeName} awansuje na poziom {level}!");
    }

        data.gold = playerData.currentGold; // Zapisujemy złoto

        if (playerData.currentEquippedItem != null)
        {
            data.equippedItemFileName = playerData.currentEquippedItem.name;
        }
        else
        {
            data.equippedItemFileName = "";
        }

        // 6. Zapis do pliku
        string json = JsonUtility.ToJson(data);
        File.WriteAllText(Application.persistentDataPath + "/savefile.json", json);

    public void LoadGame()
    {
        string path = Application.persistentDataPath + "/savefile.json";
        if (File.Exists(path))
        {
            string json = File.ReadAllText(path);
            SaveData data = JsonUtility.FromJson<SaveData>(json);

            // 1. Przywracanie statystyk
            playerData.team = new List<Pokezaur>(data.team);
            foreach (Pokezaur poke in playerData.team)
            {
                EnemyData species = allSpeciesDatabase.Find(e => e.enemyName == poke.pokeName);
                if (species != null)
                {
                    poke.sprite = species.battleSprite;
                    poke.backSprite = species.backSprite != null ? species.backSprite : species.battleSprite;
                }
            }

            playerData.collectedItemIDs = data.collectedItemIDs;
            playerData.currentGold = data.gold; // Wczytujemy złoto
            playerData.moveSpeed = data.movementSpeed;

            // 2. Przywracanie świata
            BattleTransitionManager.defeatedEnemies = data.defeatedEnemies;
            playerData.collectedItemIDs = data.collectedItems; // Żeby zebrane itemy nie wróciły

            // 3. Ustawienie flagi teleportacji
            if (data.position != null && data.position.Length == 3)
            {
                BattleTransitionManager.targetSpawnPosition = new Vector3(data.position[0], data.position[1], data.position[2]);
                BattleTransitionManager.spawnPositionActive = true;
            }

            // 4. INWENTARZ - WCZYTYWANIE Z RESOURCES
            playerData.inventory.Clear(); // Czyścimy obecny
            playerData.currentEquippedItem = null; // Resetujemy EQ

            foreach (SavedInventoryItem savedSlot in data.inventory)
            {
                // Szukamy pliku w Assets/Resources/Items/
                ItemData itemAsset = Resources.Load<ItemData>("Items/" + savedSlot.fileName);

                if (itemAsset != null)
                {
                    playerData.inventory.Add(new InventorySlot(itemAsset, savedSlot.amount));
                }
                else
                {
                    Debug.LogWarning($"Nie znaleziono pliku przedmiotu: {savedSlot.fileName}. Upewnij się, że jest w folderze Resources/Items!");
                }
            }

            // 5. PRZYWRACANIE ZAŁOŻONEGO EKWIPUNKU
            if (!string.IsNullOrEmpty(data.equippedItemFileName))
            {
                ItemData equippedItem = Resources.Load<ItemData>("Items/" + data.equippedItemFileName);
                if (equippedItem != null)
                {
                    // Ustawiamy tylko referencję (statystyki wczytaliśmy już w punkcie 1)
                    playerData.currentEquippedItem = equippedItem;
                }
            }

            // 6. Załadowanie sceny
            BattleTransitionManager.isReturningFromBattle = false;

        foreach (Transform child in shopContainer) Destroy(child.gameObject);
        foreach (Transform child in playerContainer) Destroy(child.gameObject);

        if (currentShopStock != null)
        {
            foreach (ItemData item in currentShopStock)

            }

        else
        {
            // Widok gracza: Pokaż cenę sprzedaży (np. połowa)
            texts[0].text = item.itemName;
            if (texts.Length >= 2)
            {
                // Znajdź ilość tego przedmiotu w plecaku
                InventorySlot slot = playerData.inventory.Find(x => x.item == item);
                int amount = slot != null ? slot.amount : 0;
                int sellPrice = item.price / 2;

                // Np.: "x5 (50 G)"
                texts[1].text = $"x{amount} <color=green>{sellPrice} G</color>";
            }
        }

        // Obrazek (jeśli masz ikonki w itemData)
        // Image icon = slotObj.transform.Find("Icon").GetComponent<Image>();
        // icon.sprite = item.icon;

        // --- INTERAKCJA ---
        Button btn = slotObj.GetComponent<Button>();
        btn.onClick.RemoveAllListeners();

            btn.onClick.AddListener(() => BuyItem(item));

        // Pancerne szukanie gracza w każdej klatce, dopóki go nie znajdzie
        if (player == null)
        {
            player = FindObjectOfType<PlayerMovement>();

            // Jeśli gracza nadal nie ma w tej klatce, przerywamy Update i unikamy błędu
            if (player == null) return;
        }

        // Obliczamy dystans. 'player.transform.position' jest w 100% bezpieczne.
        float distance = Vector2.Distance(transform.position, player.transform.position);

        if (distance <= interactionDistance)
        {
            if (GameNotifications.instance != null)
            {
                GameNotifications.instance.ShowMessage($"[E] Zbadaj jajko: {starterData.enemyName}", 0.5f);
            }

            if (Input.GetKeyDown(KeyCode.E))
            {
                if (StarterSelectionUI.instance != null && !StarterSelectionUI.instance.panel.activeSelf)
                {
                    StarterSelectionUI.instance.OpenPanel(starterData, professorDescription);
                }
            }
        }
    }
}

using UnityEngine;
using UnityEngine.UI;
using TMPro;

public class StarterSelectionUI : MonoBehaviour
{
    public static StarterSelectionUI instance;

    [Header("Elementy UI")]
    public GameObject panel;
    public Image pokeImage;
    public TextMeshProUGUI descriptionText;

    [Header("Dane")]
    public PlayerData playerData;

    private EnemyData currentStarterData;

    // Zmienna przechowująca skrypt ruchu gracza, by móc go wyłączyć
    private PlayerMovement playerMovementRef;

    void Awake()
    {
        // Klasyczny Singleton – upewnia się, że to okno jest tylko jedno
        if (instance == null) instance = this;
        panel.SetActive(false);
    }

    void Update()
    {
        // Jeśli okno jest widoczne, zamykamy je TYLKO klawiszem Escape (lub przyciskiem na ekranie)
        if (panel.activeSelf)
        {
            if (Input.GetKeyDown(KeyCode.Escape))
            {
                ChooseNo();
            }
        }
    }

    public void OpenPanel(EnemyData starterData, string description)
    {
        currentStarterData = starterData;
        pokeImage.sprite = starterData.battleSprite;
        descriptionText.text = description;

        panel.SetActive(true);

        // Znajdujemy gracza po tagu i unieruchamiamy go.
        GameObject player = GameObject.FindGameObjectWithTag("Player");
        if (player != null)
        {
            playerMovementRef = player.GetComponent<PlayerMovement>();
            if (playerMovementRef != null)
            {
                // To fizycznie dezaktywuje skrypt. Gracz "zamarza".
                playerMovementRef.enabled = false;

                // Na wypadek, gdyby gracz był akurat w trakcie chodzenia 
                // i zamarłby w pozycji biegu, zatrzymujemy go.
                Animator anim = player.GetComponent<Animator>();
                if (anim != null) anim.SetFloat("Speed", 0);

                Rigidbody2D rb = player.GetComponent<Rigidbody2D>();
                if (rb != null) rb.linearVelocity = Vector2.zero;
            }
        }
    }

    public void ChooseYes()
    {
        // Dodanie Pokezaura do drużyny (struktura listowa).
        Pokezaur newPoke = new Pokezaur(currentStarterData);
        playerData.team.Add(newPoke);

        // Opcjonalne powiadomienie.
        if (GameNotifications.instance != null)
        {
            GameNotifications.instance.ShowMessage($"Wybrałeś {newPoke.pokeName}!", 5f);
        }

        // Sprzątanie po stole – usuwamy wszystkie jajka oznaczone nowym tagiem.
        GameObject[] eggs = GameObject.FindGameObjectsWithTag("StarterEgg");
        foreach (GameObject egg in eggs)
        {
            Destroy(egg);
        }

        // Usunięcie blokady ucieczki, jeśli taka jest przed drzwiami
        GameObject doorBlocker = GameObject.Find("DoorBlocker"); //lub odpowiednia nazwa
        if (doorBlocker != null)
        {
            doorBlocker.SetActive(false);
        }


        ClosePanel();
    }

    public void ChooseNo()
    {
        ClosePanel();
    }

    private void ClosePanel()
    {
        panel.SetActive(false);

        // Zwracamy graczowi możliwość ruchu!
        if (playerMovementRef != null)
        {
            playerMovementRef.enabled = true;
        }
    }
}

using UnityEngine;
using TMPro;
using UnityEngine.UI;
using System.Collections.Generic;
using UnityEngine.SceneManagement;

public class StatsMenu : MonoBehaviour
{
    public static StatsMenu instance;

    [Header("Główne")]
    public GameObject menuPanel;
    public PlayerData playerData;

    [Header("Statystyki (Teksty)")]
    public TextMeshProUGUI nameText;
    public TextMeshProUGUI levelText;
    public TextMeshProUGUI hpText;
    public TextMeshProUGUI xpText;
    public TextMeshProUGUI AttackText;
    public TextMeshProUGUI DefText;
    public TextMeshProUGUI goldText;
    public TextMeshProUGUI speedText;

    [Header("Podgląd Drużyny (Nowe)")]
    public Image pokeImage;                  // Obrazek aktualnie wybranego Pokezaura
    public TextMeshProUGUI slotIndexText;    // Np. "Pokezaur 1 / 3"
    public Button setLeaderButton;           // Przycisk "Ustaw jako lidera"

    [Header("Ekwipunek")]
    public GameObject inventoryContainer;
    public GameObject itemSlotPrefab;

    public bool isMenuOpen = false;
    private int selectedPokeIndex = 0; // Którego stworka z drużyny aktualnie oglądamy (0 - 5)

    void Update()
    {
        string currentScene = SceneManager.GetActiveScene().name;
        if (currentScene == "MainMenu" || currentScene == "BattleScene")
        {
            return;
        }

        if (PauseMenu.GameIsPaused)
        {
            return;
        }

        if (ShopMenu.instance != null && ShopMenu.instance.isShopOpen)
        {
            if (Input.GetKeyDown(KeyCode.B))
            {
                ShopMenu.instance.CloseShop();
            }
            return;
        }

        if (Input.GetKeyDown(KeyCode.B))
        {
            if (isMenuOpen)
            {
                CloseMenu();
            }
            else
            {
                OpenMenu();
            }
        }
    }

    private void Awake()
    {
        if (instance == null)
        {
            instance = this;
        }
    }

    void OpenMenu()
    {
        menuPanel.SetActive(true);
        isMenuOpen = true;
        selectedPokeIndex = 0; // Po otwarciu domyślnie pokazujemy lidera (slot 0)

        Time.timeScale = 0f;

        UpdateStats();
        UpdateInventory();
    }

    public void CloseMenu()
    {
        menuPanel.SetActive(false);
        isMenuOpen = false;

        Time.timeScale = 1f;
    }

    // Podepnij pod przycisk ">" (Następny Pokezaur) w oknie StatsMenu
    public void NextPokezaur()
    {
        if (playerData.team.Count <= 1) return;
        selectedPokeIndex = (selectedPokeIndex + 1) % playerData.team.Count;
        UpdateStats();
    }

    // Podepnij pod przycisk "<" (Poprzedni Pokezaur) w oknie StatsMenu
    public void PreviousPokezaur()
    {
        if (playerData.team.Count <= 1) return;
        selectedPokeIndex--;
        if (selectedPokeIndex < 0) selectedPokeIndex = playerData.team.Count - 1;
        UpdateStats();
    }

    // Podepnij pod przycisk "Ustaw jako lidera" w oknie StatsMenu
    public void SetSelectedAsLeader()
    {
        if (selectedPokeIndex <= 0 || selectedPokeIndex >= playerData.team.Count) return;

        // Bezpieczna zamiana lidera (przekłada też założony ekwipunek na nowego lidera)
        playerData.SwapLeader(selectedPokeIndex);

        selectedPokeIndex = 0; // Teraz nasz wybrany stworek jest na pozycji 0
        UpdateStats();
    }

    void UpdateStats()
    {
        if (goldText != null) goldText.text = $"{playerData.currentGold}";
        if (speedText != null) speedText.text = $"Szybkość: {playerData.moveSpeed}";

        if (playerData.team.Count > 0)
        {
            // Zabezpieczenie zakresu indeksu
            if (selectedPokeIndex >= playerData.team.Count)
                selectedPokeIndex = 0;

            Pokezaur current = playerData.team[selectedPokeIndex];
            string leaderTag = (selectedPokeIndex == 0) ? " [LIDER]" : "";

            nameText.text = current.pokeName + leaderTag;
            levelText.text = "Poziom: " + current.level;
            hpText.text = $"HP: {current.currentHP} / {current.maxHP}";
            xpText.text = $"XP: {current.currentXP} / {current.xpToNextLevel}";
            AttackText.text = $"Siła: {current.damage}";
            DefText.text = $"Obrona: {current.defense}";

            if (pokeImage != null && current.sprite != null)
            {
                pokeImage.enabled = true;
                pokeImage.sprite = current.sprite;
            }

            if (slotIndexText != null)
            {
                slotIndexText.text = $"{selectedPokeIndex + 1} / {playerData.team.Count}";
            }

            if (setLeaderButton != null)
            {
                // Przycisk "Ustaw jako lidera" jest aktywny tylko, gdy nie oglądamy już lidera
                setLeaderButton.interactable = (selectedPokeIndex != 0);
            }
        }
        else
        {
            nameText.text = "Brak Pokezaurów";
            levelText.text = ""; hpText.text = ""; xpText.text = "";
            AttackText.text = ""; DefText.text = "";
            if (pokeImage != null) pokeImage.enabled = false;
            if (slotIndexText != null) slotIndexText.text = "0 / 0";
            if (setLeaderButton != null) setLeaderButton.interactable = false;
        }
    }

    void UpdateInventory()
    {
        foreach (Transform child in inventoryContainer.transform)
        {
            Destroy(child.gameObject);
        }

        foreach (InventorySlot slot in playerData.inventory)
        {
            GameObject newSlot = Instantiate(itemSlotPrefab, inventoryContainer.transform);
            TextMeshProUGUI[] texts = newSlot.GetComponentsInChildren<TextMeshProUGUI>();

            if (texts.Length >= 1)
            {
                if (playerData.currentEquippedItem == slot.item)
                {
                    texts[0].text = slot.item.itemName + " [Założony]";
                    texts[0].color = Color.green;
                }
                else
                {
                    texts[0].text = slot.item.itemName;
                    texts[0].color = Color.black;
                }
            }

            if (texts.Length >= 2) texts[1].text = "x" + slot.amount;

            Button btn = newSlot.GetComponent<Button>();
            if (btn != null)
            {
                btn.onClick.AddListener(() => OnItemClicked(slot.item));
            }
        }
    }

    void OnItemClicked(ItemData item)
    {
        // Używamy przedmiotu na aktualnie przeglądanym Pokezaurze (selectedPokeIndex)
        playerData.UseItem(item, selectedPokeIndex);
        UpdateStats();
        UpdateInventory();
    }
}
